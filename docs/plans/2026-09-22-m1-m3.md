- Step 1: ecrire `typecheck.py` :
  ```python
  # typecheck.py — python typecheck.py ; sortie = nb d'erreurs
  from __future__ import annotations
  import mypy, mypy.api, mypy.options, pathlib
  opt = mypy.options.Options(); opt.strict = True; opt.cache_dir = None
  files = [str(pathlib.Path(p)) for p in ("intuition.py", "eval_harness.py",
            "e2e_bench.py", "make_report.py", "export_csv.py", "run_all.py")]
  res = mypy.api.run(files, opt)
  print(f"mypy strict: {res} erreurs")
  raise SystemExit(1 if res else 0)
  ```
